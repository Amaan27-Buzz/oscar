cmake_minimum_required(VERSION 3.20)
project(oscar_edge_reference C)
set(CMAKE_C_STANDARD 11)
add_executable(edge_reference reference.c ../protocol/codec.c ../protocol/stream.c ../pico/controller.c)
target_include_directories(edge_reference PRIVATE ../protocol)
if(MSVC)
  target_compile_definitions(edge_reference PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(edge_reference PRIVATE /W4 /WX)
else()
  target_compile_options(edge_reference PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
