cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(oscar_edge_pico C CXX ASM)
set(CMAKE_C_STANDARD 11)
option(EDGE_WATCHDOG_TEST "Owner-only watchdog fault input and reset marker" OFF)
pico_sdk_init()
add_executable(oscar_edge_pico main.c controller.c ../protocol/codec.c ../protocol/stream.c)
target_include_directories(oscar_edge_pico PRIVATE ../protocol)
target_link_libraries(oscar_edge_pico pico_stdlib hardware_uart hardware_adc hardware_watchdog)
target_compile_options(oscar_edge_pico PRIVATE -Wall -Wextra -Werror)
if(EDGE_WATCHDOG_TEST)
  target_compile_definitions(oscar_edge_pico PRIVATE EDGE_WATCHDOG_TEST=1)
endif()
pico_add_extra_outputs(oscar_edge_pico)
